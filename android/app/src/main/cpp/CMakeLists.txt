cmake_minimum_required(VERSION 3.22)
project(tapscene_ocr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
option(TAPSCENE_HOST_TEST "Build the synthetic-fixture-only host runner" OFF)
get_filename_component(TAPSCENE_APP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(TAPSCENE_OCR_ROOT "${TAPSCENE_APP_DIR}/build/ocr" CACHE PATH "SHA-verified official dependencies")
set(TAPSCENE_OCR_SOURCES "${TAPSCENE_OCR_ROOT}/src" CACHE PATH "Verified source tree")
if(NOT EXISTS "${TAPSCENE_OCR_SOURCES}/opencv/CMakeLists.txt")
  message(FATAL_ERROR "Run python3 tools/ocr-native/prepare.py before configuring OCR")
endif()
if(NOT ANDROID AND NOT TAPSCENE_HOST_TEST)
  message(FATAL_ERROR "Host builds are restricted to the explicit fixture target")
endif()
if(ANDROID AND CMAKE_SYSTEM_VERSION VERSION_LESS "26")
  message(FATAL_ERROR "TapScene OCR requires Android API 26 or later")
endif()
# Every optional WITH_* backend in the pinned OpenCV root is disabled, including
# future additions to this exact source. No codecs, hardware SDK or downloaded HAL.
get_filename_component(TAPSCENE_REPO_ROOT "${TAPSCENE_APP_DIR}/../.." ABSOLUTE)
file(SHA256 "${TAPSCENE_REPO_ROOT}/tools/ocr-native/dependencies.lock.json" requested_lock)
file(SHA256 "${TAPSCENE_OCR_ROOT}/dependencies.lock.json" prepared_lock)
if(NOT requested_lock STREQUAL prepared_lock)
  message(FATAL_ERROR "Prepared OCR inputs use a different dependency lock; run prepare.py")
endif()
foreach(option WITH_ADE WITH_FREETYPE WITH_PLAIDML WITH_OAK WITH_GAPI_ONEVPL)
  set(${option} OFF CACHE BOOL "No optional module setup" FORCE)
endforeach()
file(STRINGS "${TAPSCENE_OCR_SOURCES}/opencv/CMakeLists.txt" OPENCV_OPTIONS REGEX "^OCV_OPTION\\(WITH_[A-Z0-9_]+")
foreach(line IN LISTS OPENCV_OPTIONS)
  string(REGEX MATCH "WITH_[A-Z0-9_]+" option_name "${line}")
  set(${option_name} OFF CACHE BOOL "TapScene core/imgproc-only profile" FORCE)
endforeach()
foreach(option BUILD_SHARED_LIBS BUILD_ZLIB BUILD_TIFF BUILD_OPENJPEG BUILD_JASPER BUILD_JPEG BUILD_PNG BUILD_WEBP BUILD_TBB BUILD_IPP_IW BUILD_ITT BUILD_CLAPACK BUILD_JAVA BUILD_OBJC BUILD_ANDROID_PROJECTS BUILD_ANDROID_EXAMPLES BUILD_TESTS BUILD_PERF_TESTS BUILD_EXAMPLES BUILD_DOCS BUILD_opencv_apps BUILD_opencv_gapi BUILD_opencv_world BUILD_opencv_python2 BUILD_opencv_python3 ENABLE_PRECOMPILED_HEADERS ENABLE_LTO ENABLE_THIN_LTO ENABLE_CCACHE OPENCV_ENABLE_ALLOCATOR_STATS OPENCV_ENABLE_NONFREE OPENCV_GENERATE_PKGCONFIG OPENCV_DNN_OPENCL OPENCV_DNN_CUDA OPENCV_DNN_MLAS OPENCV_TRACE ENABLE_INSTRUMENTATION)
  set(${option} OFF CACHE BOOL "TapScene minimal offline profile" FORCE)
endforeach()
set(BUILD_LIST "core,imgproc" CACHE STRING "Only raw pixel operations" FORCE)
set(CPU_DISPATCH "" CACHE STRING "No extra fat CPU variants" FORCE)
set(CV_TRACE OFF CACHE BOOL "No native trace data" FORCE)
set(OPENCV_DISABLE_FILESYSTEM_SUPPORT ON CACHE BOOL "No OpenCV filesystem API" FORCE)
set(OPENCV_DISABLE_ENV_SUPPORT ON CACHE BOOL "No environment-driven OpenCV behavior" FORCE)
set(OPENCV_SKIP_PYTHON_LOADER ON CACHE BOOL "No Python" FORCE)
set(OPENCV_SKIP_PYTHON_EXPERIMENTS ON CACHE BOOL "No Python discovery" FORCE)
set(OPENCV_SKIP_CMAKE_ROOT_CONFIG ON CACHE BOOL "Subproject" FORCE)
add_subdirectory("${TAPSCENE_OCR_SOURCES}/opencv" "opencv" EXCLUDE_FROM_ALL)
foreach(target opencv_core opencv_imgproc)
  target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections)
endforeach()
add_library(tapscene_ort SHARED IMPORTED GLOBAL)
if(ANDROID)
  set(TAPSCENE_ORT_LIBRARY "${TAPSCENE_OCR_ROOT}/ort/${ANDROID_ABI}/libonnxruntime.so")
else()
  set(TAPSCENE_ORT_LIBRARY "${TAPSCENE_OCR_ROOT}/ort/host/libonnxruntime.so")
endif()
if(NOT EXISTS "${TAPSCENE_ORT_LIBRARY}")
  message(FATAL_ERROR "Official ORT missing. Host builds require prepare.py --host")
endif()
set_target_properties(tapscene_ort PROPERTIES IMPORTED_LOCATION "${TAPSCENE_ORT_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${TAPSCENE_OCR_ROOT}/include/onnxruntime")
find_package(Threads REQUIRED)
add_library(tapscene_ocr_core STATIC ocr_core.cpp)
target_include_directories(tapscene_ocr_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE
  "${TAPSCENE_OCR_SOURCES}/opencv/modules/core/include"
  "${TAPSCENE_OCR_SOURCES}/opencv/modules/imgproc/include"
  "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(tapscene_ocr_core PUBLIC opencv_imgproc opencv_core tapscene_ort Threads::Threads)
target_compile_options(tapscene_ocr_core PRIVATE -Wall -Wextra -Werror -ffunction-sections -fdata-sections)
if(ANDROID)
  add_library(tapscene_ocr SHARED ocr_jni.cpp quiet_stdio.cpp)
  target_link_libraries(tapscene_ocr PRIVATE tapscene_ocr_core)
  target_compile_options(tapscene_ocr PRIVATE -Wall -Wextra -Werror)
  target_link_options(tapscene_ocr PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map)
  set(TAPSCENE_FINAL_TARGET tapscene_ocr)
else()
  add_executable(tapscene_ocr_fixture fixture_runner.cpp quiet_stdio.cpp)
  add_executable(tapscene_ocr_core_tests core_tests.cpp quiet_stdio.cpp)
  foreach(target tapscene_ocr_fixture tapscene_ocr_core_tests)
    target_link_libraries(${target} PRIVATE tapscene_ocr_core)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    target_link_options(${target} PRIVATE -Wl,--gc-sections)
  endforeach()
  set_target_properties(tapscene_ocr_fixture tapscene_ocr_core_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
  set(TAPSCENE_FINAL_TARGET tapscene_ocr_fixture tapscene_ocr_core_tests)
endif()
# Only static OpenCV references are wrapped; never redirect the whole process.
# Official ORT uses a discard logger and telemetry is explicitly disabled.
foreach(symbol printf vprintf fprintf vfprintf puts fputs putchar fputc fwrite)
  foreach(target IN LISTS TAPSCENE_FINAL_TARGET)
    target_link_options(${target} PRIVATE "-Wl,--wrap=${symbol}")
  endforeach()
endforeach()
